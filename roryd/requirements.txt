# RORY daemon — optional runtime dependencies.
#
# The daemon runs with NONE of these installed: it serves the brief, accepts
# typed commands, and says plainly that voice is unavailable. Install these to
# turn voice on. Pin numpy<2 — openWakeWord's default tflite path crashes under
# NumPy 2.x (verified).

numpy<2

# Always-on wake word. Ships a pretrained "hey_jarvis" model (CC BY-NC-SA 4.0,
# fine for personal use). ~5% of one core continuously.
openwakeword>=0.6.0

# Microphone capture. Needs PortAudio at the OS level:
#   macOS:   brew install portaudio
#   Debian:  sudo apt install libportaudio2
#   Windows: bundled with the wheel
sounddevice>=0.4.6

# Local speech-to-text for the command after the wake word. CPU int8.
faster-whisper>=1.0.0

# ElevenLabs voice out (optional). Without it the system voice is used.
requests>=2.31.0
